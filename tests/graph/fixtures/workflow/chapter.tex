\section{A worked chapter}

We fix the metric constant $m = 3$ (convention~M).

\begin{lemma}\label{lemma-1}
  $2 + 2 = 4$.
\end{lemma}

\begin{equation}\label{eq-2}
  m = 3.
\end{equation}

\begin{theorem}\label{thm-3}
  The constant is consistent: by Lemma~\ref{lemma-1} and Equation~\eqref{eq-2}, $m + 1 = 4$.
\end{theorem}
